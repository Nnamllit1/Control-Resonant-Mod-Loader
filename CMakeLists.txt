cmake_minimum_required(VERSION 3.24)
project(control_resonant_mod_loader VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 and MSVC are required.")
endif()
enable_language(ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
set(WASMTIME_ROOT "${CMAKE_SOURCE_DIR}/build/deps/wasmtime-v49.0.0-x86_64-windows-c-api" CACHE PATH "Wasmtime C API")
if(NOT EXISTS "${WASMTIME_ROOT}/include/wasmtime.h")
  message(FATAL_ERROR "Run python tools/fetch-deps.py first.")
endif()
add_library(wasmtime SHARED IMPORTED)
set_target_properties(wasmtime PROPERTIES IMPORTED_IMPLIB "${WASMTIME_ROOT}/lib/wasmtime.dll.lib" IMPORTED_LOCATION "${WASMTIME_ROOT}/lib/wasmtime.dll" INTERFACE_INCLUDE_DIRECTORIES "${WASMTIME_ROOT}/include")
add_library(crml_core STATIC runtime/runtime.cpp)
target_include_directories(crml_core PUBLIC runtime)
target_link_libraries(crml_core PUBLIC wasmtime)
add_executable(crml_host tools/host.cpp)
add_executable(crml_gameplay_tests tests/gameplay_host.cpp)
target_link_libraries(crml_gameplay_tests PRIVATE crml_core)
target_link_libraries(crml_host PRIVATE crml_core)
add_executable(crml_wat tools/wat.cpp)
target_link_libraries(crml_wat PRIVATE wasmtime)
add_library(crml_runtime SHARED runtime/entry.cpp)
target_link_libraries(crml_runtime PRIVATE crml_core)
option(CRML_MOVEMENT_PROBE "Build opt-in, version-gated experimental gameplay and movement diagnostics" OFF)
if(CRML_MOVEMENT_PROBE)
  enable_language(C)
  set(MINHOOK_ROOT "${CMAKE_SOURCE_DIR}/build/deps/minhook-1.3.4")
  if(NOT EXISTS "${MINHOOK_ROOT}/include/MinHook.h")
    message(FATAL_ERROR "Run python tools/fetch-deps.py --movement-probe first.")
  endif()
  add_library(crml_minhook STATIC "${MINHOOK_ROOT}/src/buffer.c" "${MINHOOK_ROOT}/src/hook.c" "${MINHOOK_ROOT}/src/trampoline.c" "${MINHOOK_ROOT}/src/hde/hde64.c")
  target_include_directories(crml_minhook PUBLIC "${MINHOOK_ROOT}/include")
  target_sources(crml_runtime PRIVATE runtime/movement_probe.cpp runtime/movement_view.cpp runtime/noclip.cpp runtime/overlay.cpp runtime/input_filter.cpp runtime/fall_guard.cpp)
  target_compile_definitions(crml_runtime PRIVATE CRML_MOVEMENT_PROBE)
  target_link_libraries(crml_runtime PRIVATE crml_minhook bcrypt user32 gdi32 d3d12 dxgi)
  add_executable(crml_probe_tests tests/test_movement_probe.cpp runtime/movement_view.cpp runtime/noclip.cpp runtime/fall_guard.cpp)
  target_include_directories(crml_probe_tests PRIVATE runtime)
  target_link_libraries(crml_probe_tests PRIVATE crml_minhook)
  add_test(NAME movement_probe COMMAND crml_probe_tests)
  add_executable(crml_input_tests tests/test_input_filter.cpp runtime/input_filter.cpp)
  target_include_directories(crml_input_tests PRIVATE runtime)
  target_link_libraries(crml_input_tests PRIVATE crml_minhook user32)
  add_test(NAME input_filter COMMAND crml_input_tests)
  add_executable(crml_overlay_tests tests/test_overlay.cpp runtime/overlay.cpp)
  target_include_directories(crml_overlay_tests PRIVATE runtime)
  target_link_libraries(crml_overlay_tests PRIVATE crml_minhook user32 gdi32 d3d12 dxgi)
  add_test(NAME overlay COMMAND crml_overlay_tests)
  set_tests_properties(overlay PROPERTIES TIMEOUT 60)
  install(FILES "${MINHOOK_ROOT}/LICENSE.txt" DESTINATION licenses/minhook)
endif()
add_library(crml_proxy SHARED loader/xinput.cpp loader/forward.asm loader/xinput.def)
set_target_properties(crml_proxy PROPERTIES OUTPUT_NAME xinput1_4)
foreach(target crml_host crml_wat crml_runtime crml_gameplay_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WASMTIME_ROOT}/lib/wasmtime.dll" "$<TARGET_FILE_DIR:${target}>")
endforeach()
install(TARGETS crml_proxy RUNTIME DESTINATION .)
install(TARGETS crml_runtime crml_host crml_wat RUNTIME DESTINATION crml)
install(FILES "${WASMTIME_ROOT}/lib/wasmtime.dll" DESTINATION crml)
install(FILES "${WASMTIME_ROOT}/LICENSE" DESTINATION licenses/wasmtime)
install(DIRECTORY sdk/ DESTINATION sdk)
install(DIRECTORY examples/ DESTINATION examples)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES tools/install.py tools/inspect-game.py tools/engine_research.py tools/engine_query.py tools/verify_engine_map.py DESTINATION tools)
install(FILES README.md THIRD_PARTY.md compatibility.json DESTINATION .)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME sandbox COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/test_runtime.py" --bin "$<TARGET_FILE_DIR:crml_host>")
add_test(NAME proxy COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/test_proxy.py" --bin "$<TARGET_FILE_DIR:crml_proxy>")
add_test(NAME installer COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/test_installer.py")
add_test(NAME engine_research COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/test_engine_research.py")
set_tests_properties(sandbox proxy installer engine_research PROPERTIES TIMEOUT 60)
